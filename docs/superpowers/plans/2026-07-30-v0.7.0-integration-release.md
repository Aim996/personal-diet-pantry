**Goal:** 把已经完成的可信操作、食物决策和 Skill 行为合并为版本一致、可升级、可回滚、可复现构建且有真实渠道证据的 v0.7.0。

**Tech Stack:** Python 3.11–3.13、pytest 8、TypeScript 5.9、Vitest 3、Node.js 22.22.3+、npm、PowerShell、OpenClaw 2026.5.17+。

## Global Constraints

- 本计划只能在 TW1–TW8、FD1–FD8、SB1–SB8 全部通过后开始。
- 版本统一为 `0.7.0`；所有版本来源必须一次更新并由测试核对。
- 发布构建必须从 clean Git commit 运行。

- OpenClaw 管理命令未知时不得编造命令；渠道操作使用目标实例实际支持的界面，仓库只校验证据文件。
- 源码包包含测试，因此敏感扫描必须检查真实归档成员。

and add the bounded stage:

```python
def _run(
    command: list[str],
    *,
    cwd: Path,
    env: dict[str, str] | None = None,
    timeout_seconds: int = 180,
    stage: str,
) -> subprocess.CompletedProcess[str]:
    try:
        return subprocess.run(
            command,
            check=True,
            cwd=cwd,
            env=env,
            capture_output=True,
            text=True,
            encoding="utf-8",
            timeout=timeout_seconds,
        )
    except subprocess.TimeoutExpired as error:
        raise RuntimeError(
            f"{stage} timed out after {timeout_seconds}s"
        ) from error
```

```python
def _run(
    command: list[str],
    *,
    cwd: Path,
    capture: bool = False,
    env: dict[str, str] | None = None,
    timeout_seconds: int = 180,
    stage: str,
) -> subprocess.CompletedProcess[str]:
    try:
        return subprocess.run(
            command,
            cwd=cwd,
            check=True,
            capture_output=capture,
            text=True,
            encoding="utf-8",
            env=env,
            timeout=timeout_seconds,
        )
    except subprocess.TimeoutExpired as error:
        raise ReleaseBuildError(
            f"{stage} timed out after {timeout_seconds}s"
        ) from error
    except (OSError, subprocess.CalledProcessError) as error:
        raise ReleaseBuildError(
            f"{stage} failed: {Path(command[0]).name}"
        ) from error
```

- [ ] **Step 2: Add RED migration-matrix tests**

Create the migration test module with these helpers and tests:

```python
from __future__ import annotations

from pathlib import Path
import shutil
import sqlite3

from personal_diet_pantry import database
from personal_diet_pantry.backup import create_backup, restore_backup
from personal_diet_pantry.maintenance_control import MaintenanceController
from personal_diet_pantry.paths import ensure_data_directories, resolve_data_paths
from personal_diet_pantry.service import DietService


PROJECT_ROOT = Path(__file__).resolve().parents[2]
SEED_SQL = PROJECT_ROOT / "tests" / "fixtures" / "v067_upgrade_seed.sql"


def _assert_healthy(connection: sqlite3.Connection) -> None:
    assert connection.execute("PRAGMA foreign_key_check").fetchall() == []
    assert connection.execute("PRAGMA integrity_check").fetchone()[0] == "ok"


def _seed_v067(data_dir: Path) -> None:
    data_dir.mkdir(parents=True)
    legacy_migrations = data_dir / "migrations-v067"
    legacy_migrations.mkdir()
    for source in sorted((PROJECT_ROOT / "migrations").glob("*.sql")):
        if int(source.name.split("_", 1)[0]) <= 18:
            shutil.copy2(source, legacy_migrations / source.name)
    connection = database.connect_database(data_dir / "diet.sqlite")
    try:
        database.apply_migrations(connection, legacy_migrations)
        connection.executescript(SEED_SQL.read_text(encoding="utf-8"))
        connection.commit()
        _assert_healthy(connection)
    finally:
        connection.close()


def _schema_version(connection: sqlite3.Connection) -> int:
    return int(
        connection.execute("SELECT max(version) FROM schema_migrations").fetchone()[0]
    )


def test_empty_database_applies_001_through_021(tmp_path: Path) -> None:
    with DietService(
        PROJECT_ROOT,
        plugin_config={"dataDir": str(tmp_path / "fresh")},
        env={},
    ) as service:
        assert _schema_version(service.connection) == 21
        _assert_healthy(service.connection)


def test_v067_fixture_upgrades_without_losing_business_rows(tmp_path: Path) -> None:
    data_dir = tmp_path / "upgrade"
    _seed_v067(data_dir)
    with DietService(
        PROJECT_ROOT,
        plugin_config={"dataDir": str(data_dir)},
        env={},
    ) as service:
        assert _schema_version(service.connection) == 21
        assert service.connection.execute(
            "SELECT count(*) FROM recipe_profiles"
        ).fetchone()[0] == 1
        assert service.connection.execute(
            "SELECT status FROM shopping_list_items WHERE id = 1"
        ).fetchone()[0] == "planned"
        assert service.connection.execute(
            "SELECT cost_minor FROM pantry_cost_allocations WHERE id = 1"
        ).fetchone()[0] == 200
        assert service.connection.execute(
            "SELECT count(*) FROM privacy_erasure_tombstones"
        ).fetchone()[0] == 1
        _assert_healthy(service.connection)


def test_v067_empty_import_transaction_becomes_non_undoable(
    tmp_path: Path,
) -> None:
    data_dir = tmp_path / "empty-import"
    _seed_v067(data_dir)
    with DietService(
        PROJECT_ROOT,
        plugin_config={"dataDir": str(data_dir)},
        env={},
    ) as service:

            SELECT undo_policy, effect_count
            FROM transactions WHERE id = 'tx-v067-empty-import'
            """
        ).fetchone()
        assert tuple(row) == ("none", 0)
        _assert_healthy(service.connection)


def test_reopening_v070_database_is_idempotent(tmp_path: Path) -> None:
    data_dir = tmp_path / "reopen"
    _seed_v067(data_dir)
    snapshots: list[tuple[tuple[int, str], ...]] = []
    for _index in range(2):
        with DietService(
            PROJECT_ROOT,
            plugin_config={"dataDir": str(data_dir)},
            env={},
        ) as service:
            snapshots.append(
                tuple(
                    service.connection.execute(
                        "SELECT version, name FROM schema_migrations ORDER BY version"
                    )
                )
            )
            _assert_healthy(service.connection)
    assert snapshots[0] == snapshots[1]


def test_preupgrade_backup_rolls_back_with_v067_migrations(
    tmp_path: Path,
) -> None:
    data_dir = tmp_path / "restore"
    _seed_v067(data_dir)
    legacy_migrations = data_dir / "migrations-v067"
    paths = resolve_data_paths({"dataDir": str(data_dir)}, {}, None)
    ensure_data_directories(paths)
    legacy = database.connect_database(paths.database)
    backup_path = create_backup(legacy, paths, label="v067")
    legacy.close()
    with DietService(
        PROJECT_ROOT,
        plugin_config={"dataDir": str(data_dir)},
        env={},
    ) as service:
        restored = restore_backup(
            service.connection,
            paths,
            backup_path,
            legacy_migrations,
            confirmed=True,
        )
        try:
            assert _schema_version(restored.connection) == 18
            assert restored.connection.execute(
                "SELECT count(*) FROM recipe_profiles"

            assert restored.connection.execute(

            ).fetchone()[0] == "pending"
            assert restored.connection.execute(
                "SELECT cost_minor FROM pantry_cost_allocations WHERE id = 1"
            ).fetchone()[0] == 200
            assert restored.connection.execute(
                "SELECT count(*) FROM privacy_erasure_tombstones"
            ).fetchone()[0] == 1
            _assert_healthy(restored.connection)
        finally:
            restored.connection.close()
```
